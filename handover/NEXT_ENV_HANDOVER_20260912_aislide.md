import os, hashlib, time
d = open(src, 'rb').read()
if os.path.exists(dst):
    os.remove(dst)                  # ★①削除が必須（⚠️ 無いと永続化されない）
open(dst, 'wb').write(d)            # ②書込（⚠️ shutil.copy は使わない｜Errno30）
assert hashlib.md5(open(dst,'rb').read()).hexdigest() == hashlib.md5(d).hexdigest()   # ③FS照合
time.sleep(40)                      # ★④永続層の反映待機
