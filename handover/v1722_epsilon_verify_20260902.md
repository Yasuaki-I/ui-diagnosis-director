        # v17.2.2（9/2 統括判定4｜案ε）: cur = cy + 31 の固定オフセットは label に

        _lbl_lines = max(-(-len(_lbl) // _lbl_cap), 1)

                 _lbl, 15, bold=True, color=txt_color,
                 height_px=_lbl_h)
